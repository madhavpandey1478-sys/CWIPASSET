var template = document.createElement("template");

template.innerHTML =
    '<style>' + CSS + '</style>' +
    '<div class="wrap">' +
      '<div class="top">' +
        '<input id="search" class="search" placeholder="Search Position ID..." aria-label="Search Position ID">' +
        '<span class="visible">Visible: <strong id="visibleCount">0</strong></span>' +
        '<div class="company-wrap">' +
          '<span class="company-label">Company</span>' +
          '<select id="companyFilter" class="company" aria-label="Filter by company">' +
            '<option value="">All Companies</option>' +
          '</select>' +
        '</div>' +
        '<div class="actions">' +
          '<button id="load">Load Data</button>' +

      '</div>' +
      '<div class="status">' +
        '<span>Total Rows: <strong id="total">0</strong></span>' +
        '<span>Selected Rows: <strong id="selected">0</strong></span>' +
        '<span>Validation: <strong id="valid">-</strong></span>' +
        '<span>Error Rows: <strong id="errors">0</strong></span>' +
      '</div>' +
    '</div>';

class CWIPAssetEntry extends HTMLElement {

    constructor() {
        super();

        this.attachShadow({mode:"open"})
            .appendChild(template.content.cloneNode(true));

        this._rows = [];
        this._selected = new Set();
        this._options = {};
        this._companyOptions = [];
        this._page = 1;
        this._pageSize = 50;
        this._search = "";
        this._company = "";
        this._lastEvent = "";
        this._valid = null;
        this._errors = [];

        this._el = {};

        [
            "search","visibleCount","companyFilter",
            "load","validate","save","clear",
            "range","prev","page","next",
            "head","body","total","selected","valid","errors"
        ].forEach((id) => {
            this._el[id] = this.shadowRoot.getElementById(id);
        });

        this._bind();
        this._render();
    }

    connectedCallback() {
        this._fire("onReady","ready");
    }

    _bind() {
        var self = this;

        this._el.search.addEventListener("input", function() {
            self._search = this.value;
            self._page = 1;
            self._render();
        });

        this._el.companyFilter.addEventListener("change", function() {
            self._company = this.value;
            self._page = 1;
            self._render();
        });

        this._el.prev.onclick = function() {
            if (self._page > 1) {
                self._page--;
                self._render();
            }
        };

        this._el.next.onclick = function() {
            if (self._page < self._pageCount()) {
                self._page++;
                self._render();
            }
        };

        this._el.load.onclick = function() {
            self._fire("onDataChange","loadData");
        };

        this._el.validate.onclick = function() {
            self._validate();
        };

        this._el.save.onclick = function() {
            self._fire("onDataChange","saveChanges|" + self._rows.length);
        };

        this._el.clear.onclick = function() {
            self._clearSelectedOrAll();
        };
    }

    _fire(name,value) {
        this._lastEvent = String(value == null ? "" : value);
        this.dispatchEvent(new Event(name));
    }

    _key(row,index) {
        if (!row.__widgetKey) {
            row.__widgetKey = "row_" + index + "_" +
                Math.random().toString(36).slice(2);
        }
        return row.__widgetKey;
    }

    _normalize(rows) {
        return rows.map(function(src) {
            var r = Object.assign({},src);

            ALL_FIELDS.forEach(function(f) {
                if (r[f] === undefined || r[f] === null) {
                    r[f] = "";
                }
            });

            r.selected = !!r.selected;
            return r;
        });
    }

    /* SAC scripting API: load the rows as a JSON string. */
    setData(raw) {
        try {
            var parsed = typeof raw === "string" ? JSON.parse(raw) : raw;
            this._rows = this._normalize(Array.isArray(parsed) ? parsed : []);
        } catch(e) {
            this._rows = [];
        }

        this._selected.clear();
        this._page = 1;
        this._valid = null;
        this._errors = [];

        this._syncCompanyOptions();
        this._render();
    }

    /* SAC scripting API: retrieve all rows including edits. */
    getData() {
        return JSON.stringify(this._rows.map(function(r) {
            var c = Object.assign({},r);
            delete c.__widgetKey;
            return c;
        }));
    }

    getLastEvent() {
        return this._lastEvent;
    }

    getSelectedData() {
        var self = this;

        return JSON.stringify(
            this._rows.filter(function(r,i) {
                return self._selected.has(self._key(r,i));
            }).map(function(r) {
                var c = Object.assign({},r);
                delete c.__widgetKey;
                return c;
            })
        );
    }

    setCellValue(rowIndex,field,value) {
        if (!this._rows[rowIndex]) return;

        this._rows[rowIndex][field] = value;
        this._rows[rowIndex].isModified = true;
        this._valid = null;
        this._render();
    }

    /* Provide dropdown options by field name. */
    setOptions(raw) {
        try {
            this._options = typeof raw === "string" ?
                JSON.parse(raw) : (raw || {});
        } catch(e) {
            this._options = {};
        }

        this._syncCompanyOptions();
        this._render();
    }

    setRowOptions(rowIndex,field,raw) {
        try {
            var opts = typeof raw === "string" ?
                JSON.parse(raw) : raw;

            this._options[field] = opts || [];
        } catch(e) {
            this._options[field] = [];
        }

        this._render();
    }

    setCompanyOptions(raw) {
        try {
            this._companyOptions = typeof raw === "string" ?
                JSON.parse(raw) : (raw || []);
        } catch(e) {
            this._companyOptions = [];
        }

        this._syncCompanyOptions();
        this._render();
    }

    _syncCompanyOptions() {
        var vals = this._companyOptions.slice();
        var self = this;

        if (!vals.length) {
            this._rows.forEach(function(r) {
                if (r.companyCode !== "" && r.companyCode != null) {
                    var exists = vals.some(function(x) {
                        return String(typeof x === "object" ? x.value : x)
                            === String(r.companyCode);
                    });

                    if (!exists) vals.push(r.companyCode);
                }
            });
        }

        var sel = this._el.companyFilter;
        var current = this._company;

        sel.innerHTML = '<option value="">All Companies</option>';

        vals.forEach(function(o) {
            var value = typeof o === "object" ? o.value : o;
            var label = typeof o === "object" ?
                (o.text !== undefined ? o.text : o.value) : o;

            var op = document.createElement("option");
            op.value = String(value);
            op.textContent = String(label);
            sel.appendChild(op);
        });

        var hasCurrent = vals.some(function(o) {
            return String(typeof o === "object" ? o.value : o)
                === String(current);
        });

        if (hasCurrent) {
            sel.value = current;
        } else {
            this._company = "";
            sel.value = "";
        }
    }

    _filtered() {
        var q = this._search.toLowerCase();
        var co = this._company;

        return this._rows.map(function(row,index) {
            return {row:row,index:index};
        }).filter(function(item) {
            var r = item.row;
            var id = String(r.employeeId || r.positionId || "").toLowerCase();
            var comp = String(r.companyCode || "");

            return (!q || id.indexOf(q) !== -1) &&
                   (!co || comp === co);
        });
    }

    _pageCount() {
        return Math.max(1,Math.ceil(this._filtered().length / this._pageSize));
    }

    _render() {
        var self = this;
        var filtered = this._filtered();
        var pages = this._pageCount();

        if (this._page > pages) this._page = pages;

        var start = (this._page - 1) * this._pageSize;
        var items = filtered.slice(start,start + this._pageSize);

        var head =
            '<th class="selhead">Sel ' +
            '<input id="selectAll" class="check" type="checkbox">' +
            '</th>';

        FIELDS.forEach(function(c) {
            head += '<th style="width:' + c.width + 'px">' +
                self._esc(c.label) + '</th>';
        });

        this._el.head.innerHTML = head;

        var allCheck = this.shadowRoot.getElementById("selectAll");

        allCheck.checked = items.length > 0 &&
            items.every(function(x) {
                return self._selected.has(self._key(x.row,x.index));
            });

        allCheck.indeterminate = items.some(function(x) {
            return self._selected.has(self._key(x.row,x.index));
        }) && !allCheck.checked;

        allCheck.onchange = function() {
            items.forEach(function(x) {
                var k = self._key(x.row,x.index);

                if (allCheck.checked) {
                    self._selected.add(k);
                } else {
                    self._selected.delete(k);
                }

                x.row.selected = allCheck.checked;
            });

            self._render();
            self._fire("onDataChange",
                "selectPage|" + self._page + "|" + allCheck.checked);
        };

        this._el.body.innerHTML = "";

        if (!items.length) {
            this._el.body.innerHTML =
                '<tr><td class="empty" colspan="' +
                (FIELDS.length + 1) +
                '">No matching records</td></tr>';
        }

        items.forEach(function(item) {
            var r = item.row;
            var idx = item.index;
            var k = self._key(r,idx);
            var tr = document.createElement("tr");

            if (self._selected.has(k)) tr.classList.add("selected");

            var td = document.createElement("td");
            td.className = "selhead";

            var chk = document.createElement("input");
            chk.type = "checkbox";
            chk.className = "check";
            chk.checked = self._selected.has(k);

            chk.onchange = function() {
                if (chk.checked) {
                    self._selected.add(k);
                } else {
                    self._selected.delete(k);
                }

                r.selected = chk.checked;
                self._render();
                self._fire("onDataChange",
                    "selectRow|" + idx + "|" + chk.checked);
            };

            td.appendChild(chk);
            tr.appendChild(td);

            FIELDS.forEach(function(c) {
                var cell = document.createElement("td");
                var control;

                if (c.type === "select") {
                    control = document.createElement("select");
                    control.className = "cell";

                    var opts = self._options[c.field] || [];
                    var blank = document.createElement("option");

                    blank.value = "";
                    blank.textContent = "";
                    control.appendChild(blank);

                    opts.forEach(function(o) {
                        var v = typeof o === "object" ? o.value : o;
                        var txt = typeof o === "object" ?
                            (o.text !== undefined ? o.text : o.value) : o;

                        var op = document.createElement("option");
                        op.value = String(v);
                        op.textContent = String(txt);
                        control.appendChild(op);
                    });

                    control.value = String(r[c.field] || "");

                    if (!opts.length && r[c.field]) {
                        var current = document.createElement("option");
                        current.value = String(r[c.field]);
                        current.textContent = String(r[c.field]);
                        control.appendChild(current);
                        control.value = String(r[c.field]);
                    }
                } else {
                    control = document.createElement("input");
                    control.className = "cell";
                    control.type = c.type === "date" ? "date" : "text";
                    control.value = r[c.field] || "";

                    if (c.readonly) control.readOnly = true;
                }

                control.addEventListener("change",function() {
                    r[c.field] = control.value;
                    r.isModified = true;
                    self._valid = null;
                    self._updateStatus();

                    self._fire("onFieldChange",
                        "fieldChange|" + idx + "|" +
                        c.field + "|" + control.value);
                });

                cell.appendChild(control);
                tr.appendChild(cell);
            });

            self._el.body.appendChild(tr);
        });

        this._el.visibleCount.textContent = filtered.length;

        this._el.range.textContent = filtered.length ?
            "Showing " + (start + 1) + " to " +
            Math.min(start + this._pageSize,filtered.length) +
            " of " + filtered.length :
            "Showing 0 to 0 of 0";

        this._el.page.textContent = "Page " + this._page + " of " + pages;
        this._el.prev.disabled = this._page <= 1;
        this._el.next.disabled = this._page >= pages;

        this._updateStatus();
    }

    _updateStatus() {
        this._el.total.textContent = this._rows.length;
        this._el.selected.textContent = this._selected.size;
        this._el.valid.textContent =
            this._valid === null ? "-" : (this._valid ? "Valid" : "Invalid");
        this._el.errors.textContent = this._errors.length;
    }

    _validate() {
        var mandatory = [
            "companyCode","division","department",
            "costCenter","jobCode","positionTitle"
        ];

        var errors = [];

        this._rows.forEach(function(r,i) {
            var has = ALL_FIELDS.some(function(f) {
                return r[f] !== "" && r[f] !== null && r[f] !== undefined;
            });

            if (has) {
                var missing = mandatory.filter(function(f) {
                    return !r[f];
                });

                if (missing.length) {
                    errors.push({row:i,fields:missing});
                }
            }
        });

        this._errors = errors;
        this._valid = errors.length === 0;
        this._updateStatus();

        this._fire("onValidate",
            (this._valid ? "VALID" : "INVALID") + "|" + errors.length);
    }

    _clearSelectedOrAll() {
        var self = this;

        if (this._selected.size) {
            this._rows.forEach(function(r,i) {
                if (self._selected.has(self._key(r,i))) {
                    ALL_FIELDS.forEach(function(f) {
                        if (f !== "employeeId") r[f] = "";
                    });

                    r.isModified = true;
                }
            });

            this._selected.clear();
        } else {
            this._rows.forEach(function(r) {
                ALL_FIELDS.forEach(function(f) {
                    if (f !== "employeeId") r[f] = "";
                });

                r.isModified = true;
            });
        }

        this._valid = null;
        this._errors = [];
        this._render();
        this._fire("onClear","clear");
    }

    _esc(v) {
        return String(v == null ? "" : v)
            .replace(/&/g,"&amp;")
            .replace(/</g,"&lt;")
            .replace(/>/g,"&gt;")
            .replace(/"/g,"&quot;")
            .replace(/'/g,"&#039;");
    }
}

if (!customElements.get("com-madhav-cwipassetentry")) {

        "com-madhav-cwipassetentry",
        CWIPAssetEntry
    );
}
