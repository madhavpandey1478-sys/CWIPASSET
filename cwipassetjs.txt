    /* =========================================================

    class PositionManage extends HTMLElement {

        constructor() {
            super();
            this.attachShadow({ mode: "open" });
            this.shadowRoot.appendChild(tmpl.content.cloneNode(true));

            this._rows       = [];     /* full data set                          */
            this._opts       = {};     /* _opts[globalRowIndex][field] = [...]   */
            this._valid      = null;   /* null | true | false                    */
            this._lastEvent  = "";
            this._search     = "";
            this._page       = 0;      /* 0-based current page                   */
            this._filtered   = [];     /* indices into this._rows after search   */

            this._bindStatic();
            this._buildHead();
            this._refresh();
        }

        connectedCallback() {
            var self = this;
            this._fire("onReady", "ready");
        }

        /* =========================================================
           STATIC BINDINGS (called once)
           ========================================================= */

        _bindStatic() {
            var self = this;

            this.shadowRoot.getElementById("searchInp").oninput = function (e) {
                self._search = e.target.value;
                self._page   = 0;
                self._refilter();
                self._buildBody();
                self._updatePageBar();
                self._updateStatus();
            };

            this.shadowRoot.getElementById("btnLoad").onclick  = function () { self._fire("onDataChange", "loadData"); };
            this.shadowRoot.getElementById("btnVal").onclick   = function () { self._validate(); };
            this.shadowRoot.getElementById("btnSave").onclick  = function () { self._fire("onDataChange", "saveChanges|" + self._rows.length); };
            this.shadowRoot.getElementById("btnClr").onclick   = function () { self._clearSelected(); };

            this.shadowRoot.getElementById("btnPrev").onclick  = function () {
                if (self._page > 0) { self._page--; self._buildBody(); self._updatePageBar(); }
            };
            this.shadowRoot.getElementById("btnNext").onclick  = function () {
                var maxPage = Math.max(0, Math.ceil(self._filtered.length / PAGE_SIZE) - 1);
                if (self._page < maxPage) { self._page++; self._buildBody(); self._updatePageBar(); }
            };
        }

        /* =========================================================
           HEAD (built once; never changes)
           ========================================================= */

        _buildHead() {
            var self   = this;
            var thead  = this.shadowRoot.getElementById("thead");
            thead.innerHTML = "";

            var thChk = document.createElement("th");
            thChk.className = "cc";
            var chkAll = document.createElement("input");
            chkAll.id = "chkAll"; chkAll.type = "checkbox"; chkAll.className = "chk";
            chkAll.title = "Select all visible";
            chkAll.onchange = function (e) {
                var chk = e.target.checked;
                /* select / deselect only visible (filtered) page rows */
                for (var i = 0; i < self._filtered.length; i++) {
                    self._rows[self._filtered[i]].selected = chk;
                }
                self._buildBody();
                self._updateStatus();
                self._fire("onDataChange", "selectAll|" + chk);
            };
            thChk.appendChild(chkAll);
            thead.appendChild(thChk);

            for (var i = 0; i < COLS.length; i++) {
                var th = document.createElement("th");
                th.style.width = COLS[i].width + "px";
                th.textContent = COLS[i].label;
                thead.appendChild(th);
            }
        }

        /* =========================================================
           FILTERING & PAGINATION
           ========================================================= */

        _refilter() {
            var q = this._search.toLowerCase().trim();
            this._filtered = [];
            for (var i = 0; i < this._rows.length; i++) {
                if (!q || String(this._rows[i].positionId || "").toLowerCase().indexOf(q) !== -1) {
                    this._filtered.push(i);
                }
            }
        }

        _pageSlice() {
            var start = this._page * PAGE_SIZE;
            return this._filtered.slice(start, start + PAGE_SIZE);
        }

        /* =========================================================
           BODY (rebuilt on every state change)
           ========================================================= */

        _buildBody() {
            var self  = this;
            var tbody = this.shadowRoot.getElementById("tbody");
            tbody.innerHTML = "";

            var pageIdx = this._pageSlice(); /* global row indices for this page */

            for (var pi = 0; pi < pageIdx.length; pi++) {
                var gi  = pageIdx[pi];          /* global index */
                var row = this._rows[gi];
                var miss = (this._valid === false && this._hasData(row))
                    ? this._missing(row) : [];

                var tr = document.createElement("tr");
                if (row.selected) { tr.classList.add("sel"); }
                if (miss.length)  { tr.classList.add("err"); }

                /* checkbox */
                var chkTd = document.createElement("td");
                chkTd.className = "cc";
                var chk = document.createElement("input");
                chk.type = "checkbox"; chk.className = "chk";
                chk.checked = !!row.selected;

                chk.onchange = (function (globalIdx, trEl) {
                    return function (e) {
                        self._rows[globalIdx].selected = e.target.checked;
                        if (e.target.checked) { trEl.classList.add("sel"); }
                        else                  { trEl.classList.remove("sel"); }
                        self._updateStatus();
                        self._updateChkAll();
                        self._fire("onDataChange", "selectRow|" + globalIdx + "|" + e.target.checked);
                    };
                }(gi, tr));

                chkTd.appendChild(chk);
                tr.appendChild(chkTd);

                /* data cells */
                for (var ci = 0; ci < COLS.length; ci++) {
                    var col    = COLS[ci];
                    var hasErr = (miss.indexOf(col.field) !== -1);
                    var td     = document.createElement("td");

                    if (col.readonly) {
                        td.appendChild(this._buildReadonly(row, col));
                    } else if (col.type === "select") {
                        td.appendChild(this._buildSelect(gi, row, col, hasErr));
                    } else {
                        td.appendChild(this._buildInput(gi, row, col, hasErr));
                    }

                    if (hasErr) {
                        var ep = document.createElement("div");
                        ep.className = "fe";
                        ep.textContent = col.label + " is required";
                        td.appendChild(ep);
                    }

                    tr.appendChild(td);
                }

                tbody.appendChild(tr);
            }

            this._updateChkAll();
        }

        /* ── readonly text ── */
        _buildReadonly(row, col) {
            var inp = document.createElement("input");
            inp.className = "rid";
            inp.type = "text";
            inp.value = row[col.field] || "";
            inp.readOnly = true;
            inp.tabIndex = -1;
            return inp;
        }

        /* ── native select ── */
        _buildSelect(gi, row, col, hasErr) {
            var self  = this;
            var field = col.field;
            var opts  = this._getOpts(gi, field);
            var curVal = row[field] || "";

            var wrap = document.createElement("div");
            wrap.className = "sw";

            var sel = document.createElement("select");
            sel.className = hasErr ? "ce" : "";

            /* blank option */
            var blank = document.createElement("option");
            blank.value = ""; blank.textContent = "Select";
            sel.appendChild(blank);

            for (var i = 0; i < opts.length; i++) {
                var ov = typeof opts[i] === "object" ? opts[i].value : opts[i];
                var ot = typeof opts[i] === "object" ? (opts[i].text || opts[i].value) : opts[i];
                var opt = document.createElement("option");
                opt.value = String(ov);
                opt.textContent = String(ot);
                if (String(ov) === String(curVal)) { opt.selected = true; }
                sel.appendChild(opt);
            }

            sel.onchange = (function (globalIdx, f) {
                return function () {
                    self._rows[globalIdx][f]           = sel.value;
                    self._rows[globalIdx].isModified   = true;
                    self._valid                        = null;
                    self._updateStatus();
                    self._fire("onDataChange", "fieldChange|" + globalIdx + "|" + f + "|" + sel.value);
                };
            }(gi, field));

            wrap.appendChild(sel);
            return wrap;
        }

        /* ── plain text / date input ── */
        _buildInput(gi, row, col, hasErr) {
            var self  = this;
            var field = col.field;
            var inp   = document.createElement("input");
            inp.className = "ti" + (hasErr ? " ce" : "");
            inp.type  = col.type || "text";
            inp.value = row[field] || "";

            inp.onchange = (function (globalIdx, f) {
                return function () {
                    self._rows[globalIdx][f]         = inp.value;
                    self._rows[globalIdx].isModified = true;
                    self._valid                      = null;
                    self._updateStatus();
                    self._fire("onDataChange", "fieldChange|" + globalIdx + "|" + f + "|" + inp.value);
                };
            }(gi, field));

            return inp;
        }

        /* =========================================================
           VALIDATION
           ========================================================= */

        _validate() {
            var errs = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (!this._hasData(this._rows[i])) { continue; }
                if (this._missing(this._rows[i]).length > 0) { errs++; }
            }
            this._valid = (errs === 0);
            this._buildBody();
            this._updateStatus();
            this._fire("onValidate", errs === 0 ? "VALID|0" : "INVALID|" + errs);
        }

        _hasData(row) {
            for (var i = 0; i < ALL_FIELDS.length; i++) {
                var v = row[ALL_FIELDS[i]];
                if (v !== "" && v !== null && v !== undefined) { return true; }
            }
            return false;
        }

        _missing(row) {
            var m = [];
            for (var i = 0; i < MANDATORY.length; i++) {
                if (!row[MANDATORY[i]]) { m.push(MANDATORY[i]); }
            }
            return m;
        }

        _errCount() {
            var n = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (!this._hasData(this._rows[i])) { continue; }
                if (this._missing(this._rows[i]).length > 0) { n++; }
            }
            return n;
        }

        /* =========================================================
           CLEAR SELECTED
           ========================================================= */

        _clearSelected() {
            var n = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (!this._rows[i].selected) { continue; }
                /* clear only editable fields */
                for (var ci = 0; ci < COLS.length; ci++) {
                    if (!COLS[ci].readonly) { this._rows[i][COLS[ci].field] = ""; }
                }
                this._rows[i].isModified = true;
                this._rows[i].selected   = false;
                n++;
            }
            if (!n) { return; }
            this._valid = null;
            this._buildBody();
            this._updateStatus();
            this._fire("onDataChange", "clear|" + n);
        }

        /* =========================================================
           STATUS / PAGE BAR / CHK-ALL
           ========================================================= */

        _selCount() {
            var n = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (this._rows[i].selected) { n++; }
            }
            return n;
        }

        _updateStatus() {
            var sel = this._selCount();
            this.shadowRoot.getElementById("sTot").textContent  = this._rows.length;
            this.shadowRoot.getElementById("sSel").textContent  = sel;
            this.shadowRoot.getElementById("visCount").textContent = this._filtered.length;

            var valEl = this.shadowRoot.getElementById("sVal");
            var ico   = this.shadowRoot.getElementById("sIco");
            var errEl = this.shadowRoot.getElementById("sErr");
            valEl.className = "sv";

            if (this._valid === true) {
                valEl.textContent = "true";  valEl.classList.add("ok");
                errEl.textContent = "0";
                ico.innerHTML = '<svg width="12" height="12" viewBox="0 0 20 20" fill="none"><circle cx="10" cy="10" r="9" stroke="#107e3e" stroke-width="1.6"/><path d="M6 10.3l2.6 2.6L14 7.5" stroke="#107e3e" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/></svg>';
            } else if (this._valid === false) {
                valEl.textContent = "false"; valEl.classList.add("bad");
                errEl.textContent = String(this._errCount());
                ico.innerHTML = '<svg width="12" height="12" viewBox="0 0 20 20" fill="none"><circle cx="10" cy="10" r="9" stroke="#bb0000" stroke-width="1.6"/><path d="M10 6v5" stroke="#bb0000" stroke-width="1.8" stroke-linecap="round"/><circle cx="10" cy="13.6" r="1" fill="#bb0000"/></svg>';
            } else {
                valEl.textContent = "-"; errEl.textContent = "0"; ico.innerHTML = "";
            }
        }

        _updatePageBar() {
            var total = this._filtered.length;
            var pages = Math.max(1, Math.ceil(total / PAGE_SIZE));
            var start = total === 0 ? 0 : this._page * PAGE_SIZE + 1;
            var end   = Math.min((this._page + 1) * PAGE_SIZE, total);

            this.shadowRoot.getElementById("pageLabel").textContent =
                "Showing " + start + " to " + end + " of " + total +
                " \u2014 Page " + (this._page + 1) + " of " + pages;

            this.shadowRoot.getElementById("btnPrev").disabled = (this._page === 0);
            this.shadowRoot.getElementById("btnNext").disabled = (this._page >= pages - 1);
        }

        _updateChkAll() {
            var ca = this.shadowRoot.getElementById("chkAll");
            if (!ca) { return; }
            var pageIdx = this._pageSlice();
            if (!pageIdx.length) { ca.checked = false; ca.indeterminate = false; return; }
            var selOnPage = 0;
            for (var i = 0; i < pageIdx.length; i++) {
                if (this._rows[pageIdx[i]].selected) { selOnPage++; }
            }
            ca.checked       = selOnPage === pageIdx.length;
            ca.indeterminate = selOnPage > 0 && selOnPage < pageIdx.length;
        }

        /* ── full refresh (called after data changes) ── */
        _refresh() {
            this._refilter();
            this._buildBody();
            this._updatePageBar();
            this._updateStatus();
        }

        /* =========================================================
           OPTION LOOKUP
           ========================================================= */

        _getOpts(gi, field) {
            return (this._opts[gi] && this._opts[gi][field]) ? this._opts[gi][field] : [];
        }

        /* =========================================================
           EVENT EMITTER
           ========================================================= */

        _fire(name, value) {
            this._lastEvent = String(value);
            this.dispatchEvent(new Event(name));
        }

        /* =========================================================
           PUBLIC API
           ========================================================= */

        getLastEvent() { return this._lastEvent; }

        /* Replace all rows - call inside onDataChange after "loadData" */
        getData() { return JSON.stringify(this._rows); }

        setData(raw) {
            try {
                var p = typeof raw === "string" ? JSON.parse(raw) : raw;
                this._rows = Array.isArray(p) ? p : [];
            } catch (e) { this._rows = []; }
            this._valid = null;
            this._page  = 0;
            this._search = "";
            var si = this.shadowRoot.getElementById("searchInp");
            if (si) { si.value = ""; }
            this._refresh();
        }

        /* Set company label text shown in the pill */
        setCompanyLabel(text) {
            var el = this.shadowRoot.getElementById("companyLabel");
            if (el) { el.textContent = text || "Company"; }
        }

        /* Set one cell value */
        setCellValue(gi, field, val) {
            if (!this._rows[gi]) { return; }
            this._rows[gi][field]       = val;
            this._rows[gi].isModified   = true;
            this._valid = null;
            this._buildBody();
            this._updateStatus();
        }

        /* Set dropdown options for one cell (by global row index) */
        setRowOptions(gi, field, raw) {
            if (!this._opts[gi]) { this._opts[gi] = {}; }
            try {
                this._opts[gi][field] = typeof raw === "string" ? JSON.parse(raw) : raw;
            } catch (e) { this._opts[gi][field] = []; }
            this._buildBody();
        }
    }

    if (!customElements.get("com-madhav-positionmanage")) {
        customElements.define("com-madhav-positionmanage", PositionManage);
    }

})();
