    /* =========================================================
       COMPONENT
       ========================================================= */
    class CWIPAssetEntry extends HTMLElement {

        constructor() {
            super();
            this.attachShadow({ mode: "open" });
            this.shadowRoot.appendChild(tmpl.content.cloneNode(true));

            this._lastEvent = "";
            this._dd        = null;   /* { row, field } */
            this._ddSearch  = "";

            /* data store */
            this._rows = [];
            this._opts = {};   /* _opts[rowIndex][fieldName] = [...] */
            this._valid = null;

            this._buildToolbar();
            this._buildHead();
            this._buildBody();
            this._updateStatus();
        }

        connectedCallback() {
            var self = this;
            this._globalClose = function () {
                if (self._dd) {
                    self._dd = null;
                    self._ddSearch = "";
                    self._buildBody();
                }
            };
            document.addEventListener("click", this._globalClose);
            this._fire("onReady", "ready");
        }

        disconnectedCallback() {
            if (this._globalClose) {
                document.removeEventListener("click", this._globalClose);
            }
        }

        /* =========================================================
           INTERNAL HELPERS
           ========================================================= */

        _emptyRow() {
            var r = { selected: false, isModified: false };
            for (var i = 0; i < ALL_FIELDS.length; i++) { r[ALL_FIELDS[i]] = ""; }
            return r;
        }

        _fire(name, value) {
            this._lastEvent = String(value);
            this.dispatchEvent(new Event(name));
        }

        _oval(o)  { return typeof o === "object" ? o.value : o; }
        _otxt(o)  { return typeof o === "object" ? (o.text !== undefined ? o.text : o.value) : o; }

        _otext(opts, val) {
            if (!val) { return ""; }
            for (var i = 0; i < opts.length; i++) {
                if (String(this._oval(opts[i])) === String(val)) { return this._otxt(opts[i]); }
            }
            return String(val);
        }

        _ofilt(opts, q) {
            if (!q) { return opts; }
            var n = String(q).toLowerCase();
            var r = [];
            for (var i = 0; i < opts.length; i++) {
                if (String(this._otxt(opts[i])).toLowerCase().indexOf(n) !== -1 ||
                    String(this._oval(opts[i])).toLowerCase().indexOf(n) !== -1) {
                    r.push(opts[i]);
                }
            }
            return r;
        }

        _getOpts(ri, field) {
            return (this._opts[ri] && this._opts[ri][field]) ? this._opts[ri][field] : [];
        }

        _hasData(row) {
            for (var i = 0; i < ALL_FIELDS.length; i++) {
                var v = row[ALL_FIELDS[i]];
                if (v !== "" && v !== null && v !== undefined) { return true; }
            }
            return false;
        }

        _missing(row) {
            var m = [];
            for (var i = 0; i < MANDATORY.length; i++) {
                if (!row[MANDATORY[i]]) { m.push(MANDATORY[i]); }
            }
            return m;
        }

        _errCount() {
            var n = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (!this._hasData(this._rows[i])) { continue; }
                if (this._missing(this._rows[i]).length > 0) { n++; }
            }
            return n;
        }

        _selCount() {
            var n = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (this._rows[i].selected) { n++; }
            }
            return n;
        }

        /* =========================================================
           TOOLBAR
           ========================================================= */

        _buildToolbar() {
            var self = this;
            var tb = this.shadowRoot.getElementById("toolbar");

            tb.innerHTML =
                '<button id="btnDel" class="del" style="display:none">' + ICON_DEL  + ' Delete Selected</button>' +
                '<button id="btnVal">'                                   + ICON_CHECK + ' Validate</button>'        +
                '<button id="btnSave" class="primary">'                  + ICON_SAVE  + ' Save Changes</button>'   +
                '<button id="btnClr">'                                   + ICON_CLEAR + ' Clear</button>'          +
                '<button id="btnLoad">'                                  + ICON_LOAD  + ' Load Data</button>';

            this.shadowRoot.getElementById("btnDel").onclick  = function () { self._deleteRows(); };
            this.shadowRoot.getElementById("btnVal").onclick  = function () { self._validate(); };
            this.shadowRoot.getElementById("btnSave").onclick = function () { self._fire("onDataChange", "saveChanges|" + self._rows.length); };
            this.shadowRoot.getElementById("btnClr").onclick  = function () { self._clearRows(); };
            this.shadowRoot.getElementById("btnLoad").onclick = function () { self._fire("onDataChange", "loadData"); };
        }

        /* =========================================================
           TABLE HEAD
           ========================================================= */

        _buildHead() {
            var self = this;
            var tr = this.shadowRoot.getElementById("thead");
            tr.innerHTML = "";

            /* select-all checkbox */
            var thChk = document.createElement("th");
            thChk.className = "chk-cell";
            var chkAll = document.createElement("input");
            chkAll.type = "checkbox";
            chkAll.className = "chk";
            chkAll.id = "chkAll";
            chkAll.onchange = function (e) {
                var chk = e.target.checked;
                for (var i = 0; i < self._rows.length; i++) { self._rows[i].selected = chk; }
                self._buildBody();
                self._updateStatus();
                self._fire("onDataChange", "selectAll|" + chk);
            };
            thChk.appendChild(chkAll);
            tr.appendChild(thChk);

            /* column headers */
            for (var i = 0; i < COLS.length; i++) {
                var th = document.createElement("th");
                th.style.width = COLS[i].width + "px";
                th.textContent = COLS[i].label;
                tr.appendChild(th);
            }
        }

        /* =========================================================
           TABLE BODY  (only method called after data changes)
           ========================================================= */

        _buildBody() {
            var self   = this;
            var tbody  = this.shadowRoot.getElementById("tbody");
            tbody.innerHTML = "";

            for (var ri = 0; ri < this._rows.length; ri++) {
                tbody.appendChild(this._buildRow(ri));
            }

            this._updateDelBtn();
            this._updateChkAll();
        }

        _buildRow(ri) {
            var self = this;
            var row  = this._rows[ri];
            var miss = (this._valid === false && this._hasData(row)) ? this._missing(row) : [];

            var tr = document.createElement("tr");
            if (row.selected) { tr.classList.add("sel"); }
            if (miss.length)  { tr.classList.add("err"); }

            /* ── checkbox ── */
            var chkTd = document.createElement("td");
            chkTd.className = "chk-cell";
            var chk = document.createElement("input");
            chk.type = "checkbox";
            chk.className = "chk";
            chk.checked = !!row.selected;

            chk.onchange = (function (rowIdx, trEl) {
                return function (e) {
                    self._rows[rowIdx].selected = e.target.checked;
                    if (e.target.checked) { trEl.classList.add("sel"); }
                    else                  { trEl.classList.remove("sel"); }
                    self._updateDelBtn();
                    self._updateChkAll();
                    self._updateStatus();
                    self._fire("onDataChange", "selectRow|" + rowIdx + "|" + e.target.checked);
                };
            }(ri, tr));

            chkTd.appendChild(chk);
            tr.appendChild(chkTd);

            /* ── data cells ── */
            for (var ci = 0; ci < COLS.length; ci++) {
                var col    = COLS[ci];
                var hasErr = (miss.indexOf(col.field) !== -1);
                var td     = document.createElement("td");

                if (col.type === "select") {
                    td.appendChild(this._buildCombo(ri, col, hasErr));
                } else {
                    td.appendChild(this._buildInput(ri, col, hasErr));
                }

                if (hasErr) {
                    var ep = document.createElement("div");
                    ep.className   = "field-error";
                    ep.textContent = col.label + " is required";
                    td.appendChild(ep);
                }

                tr.appendChild(td);
            }

            return tr;
        }

        /* ── plain input ── */

        _buildInput(ri, col, hasErr) {
            var self = this;
            var inp  = document.createElement("input");
            inp.className = "cell" + (col.readonly ? " ro" : "") + (hasErr ? " ce" : "");
            inp.type      = col.type || "text";
            inp.value     = this._rows[ri][col.field] || "";
            if (col.readonly) { inp.readOnly = true; return inp; }

            inp.onchange = function () {
                self._rows[ri][col.field]   = inp.value;
                self._rows[ri].isModified   = true;
                self._valid                 = null;
                self._updateStatus();
                self._fire("onDataChange", "fieldChange|" + ri + "|" + col.field + "|" + inp.value);
            };
            return inp;
        }

        /* ── searchable dropdown ── */

        _buildCombo(ri, col, hasErr) {
            var self    = this;
            var field   = col.field;
            var opts    = this._getOpts(ri, field);
            var curVal  = this._rows[ri][field] || "";
            var curTxt  = this._otext(opts, curVal) || "";
            var isOpen  = !!(this._dd && this._dd.row === ri && this._dd.field === field);

            /* wrapper div */
            var wrap = document.createElement("div");
            wrap.className = "cw";

            /* toggle button */
            var btn  = document.createElement("button");
            btn.type = "button";
            btn.className = "ct" + (isOpen ? " open" : "") + (hasErr ? " ce" : "");

            var lbl = document.createElement("span");
            lbl.className  = "ct-text" + (curTxt ? "" : " ph");
            lbl.textContent = curTxt || "Select";

            var chev = document.createElement("span");
            chev.className = "ct-chev";
            chev.innerHTML = '<svg width="12" height="12" viewBox="0 0 20 20" fill="none"><path d="M5 7.5l5 5 5-5" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/></svg>';

            btn.appendChild(lbl);
            btn.appendChild(chev);

            btn.onclick = function (e) {
                e.stopPropagation();
                if (isOpen) {
                    self._dd = null;
                } else {
                    self._dd = { row: ri, field: field };
                    self._ddSearch = "";
                }
                self._buildBody();
            };

            wrap.appendChild(btn);

            /* panel (only when open) */
            if (isOpen) {
                var panel = document.createElement("div");
                panel.className = "cp";
                panel.onclick = function (e) { e.stopPropagation(); };

                /* search bar */
                var sRow = document.createElement("div");
                sRow.className = "cs";

                var sIco = document.createElement("span");
                sIco.className = "cs-icon";
                sIco.innerHTML = '<svg width="14" height="14" viewBox="0 0 20 20" fill="none"><circle cx="8.5" cy="8.5" r="5.5" stroke="currentColor" stroke-width="1.6"/><path d="M13 13l4 4" stroke="currentColor" stroke-width="1.6" stroke-linecap="round"/></svg>';

                var sInp = document.createElement("input");
                sInp.type        = "text";
                sInp.className   = "cs-inp";
                sInp.placeholder = "Search";
                sInp.value       = this._ddSearch;
                sInp.autocomplete = "off";

                sRow.appendChild(sIco);
                sRow.appendChild(sInp);

                if (this._ddSearch) {
                    var sClr = document.createElement("button");
                    sClr.type      = "button";
                    sClr.className = "cs-clr";
                    sClr.innerHTML = "&times;";
                    sClr.onclick   = function (e) {
                        e.stopPropagation();
                        self._ddSearch = "";
                        self._buildBody();
                    };
                    sRow.appendChild(sClr);
                }

                panel.appendChild(sRow);

                /* options */
                var list     = document.createElement("div");
                list.className = "co-list";
                var filtered = this._ofilt(opts, this._ddSearch);

                if (!filtered.length) {
                    var emp = document.createElement("div");
                    emp.className   = "co-empty";
                    emp.textContent = "No matches";
                    list.appendChild(emp);
                } else {
                    for (var oi = 0; oi < filtered.length; oi++) {
                        var ov   = String(this._oval(filtered[oi]));
                        var ot   = this._otxt(filtered[oi]);
                        var item = document.createElement("div");
                        item.className  = "co-item" + (ov === String(curVal) ? " picked" : "");
                        item.textContent = ot;

                        item.onclick = (function (optVal) {
                            return function (e) {
                                e.stopPropagation();
                                self._rows[ri][field]       = optVal;
                                self._rows[ri].isModified   = true;
                                self._valid                 = null;
                                self._dd                    = null;
                                self._ddSearch              = "";
                                self._buildBody();
                                self._updateStatus();
                                self._fire("onDataChange", "fieldChange|" + ri + "|" + field + "|" + optVal);
                            };
                        }(ov));

                        list.appendChild(item);
                    }
                }

                panel.appendChild(list);
                wrap.appendChild(panel);

                /* auto-focus search */
                setTimeout(function () { sInp.focus(); }, 0);

                sInp.oninput = function () {
                    self._ddSearch = sInp.value;
                    self._buildBody();
                };

                sInp.onkeydown = function (e) {
                    if (e.key === "Escape") {
                        self._dd       = null;
                        self._ddSearch = "";
                        self._buildBody();
                    }
                };
            }

            return wrap;
        }

        /* =========================================================
           ROW OPERATIONS
           ========================================================= */

        _deleteRows() {
            var kept = [], del = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (this._rows[i].selected) { del++; } else { kept.push(this._rows[i]); }
            }
            if (!del) { return; }
            this._rows = kept;
            this._valid = null;
            this._buildBody();
            this._updateStatus();
            this._fire("onDataChange", "deleteRow|" + del);
        }

        _clearRows() {
            var n = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (!this._rows[i].selected) { continue; }
                /* only clear the editable fields — keep assetId */
                this._rows[i].assetClass       = "";
                this._rows[i].capitalizedMonth = "";
                this._rows[i].isModified       = true;
                this._rows[i].selected         = false;
                n++;
            }
            if (!n) { return; }
            this._valid = null;
            this._buildBody();
            this._updateStatus();
            this._fire("onDataChange", "clear|" + n);
        }

        _validate() {
            var errs = 0;
            for (var i = 0; i < this._rows.length; i++) {
                if (!this._hasData(this._rows[i])) { continue; }
                if (this._missing(this._rows[i]).length > 0) { errs++; }
            }
            this._valid = (errs === 0);
            this._buildBody();
            this._updateStatus();
            if (errs === 0) {
                this._fire("onValidate", "VALID|0");
            } else {
                this._fire("onValidate", "INVALID|" + errs);
            }
        }

        /* =========================================================
           STATUS / COUNTS
           ========================================================= */

        _updateStatus() {
            var sel  = this._selCount();
            this.shadowRoot.getElementById("sRows").textContent = this._rows.length;
            this.shadowRoot.getElementById("sSel").textContent  = sel;

            var valEl = this.shadowRoot.getElementById("sVal");
            var ico   = this.shadowRoot.getElementById("sIco");
            var errEl = this.shadowRoot.getElementById("sErr");

            valEl.className = "sv";

            if (this._valid === true) {
                valEl.textContent = "true";  valEl.classList.add("ok");
                errEl.textContent = "0";
                ico.innerHTML = '<svg width="12" height="12" viewBox="0 0 20 20" fill="none"><circle cx="10" cy="10" r="9" stroke="#107e3e" stroke-width="1.6"/><path d="M6 10.3l2.6 2.6L14 7.5" stroke="#107e3e" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/></svg>';
            } else if (this._valid === false) {
                valEl.textContent = "false"; valEl.classList.add("bad");
                errEl.textContent = String(this._errCount());
                ico.innerHTML = '<svg width="12" height="12" viewBox="0 0 20 20" fill="none"><circle cx="10" cy="10" r="9" stroke="#bb0000" stroke-width="1.6"/><path d="M10 6v5" stroke="#bb0000" stroke-width="1.8" stroke-linecap="round"/><circle cx="10" cy="13.6" r="1" fill="#bb0000"/></svg>';
            } else {
                valEl.textContent = "-";
                errEl.textContent = "0";
                ico.innerHTML = "";
            }
        }

        _updateDelBtn() {
            var el = this.shadowRoot.getElementById("btnDel");
            if (el) { el.style.display = this._selCount() > 0 ? "inline-flex" : "none"; }
        }

        _updateChkAll() {
            var ca = this.shadowRoot.getElementById("chkAll");
            if (!ca) { return; }
            var sel = this._selCount();
            ca.checked       = (this._rows.length > 0 && sel === this._rows.length);
            ca.indeterminate = (sel > 0 && sel < this._rows.length);
        }

        /* =========================================================
           PUBLIC API
           ========================================================= */

        getLastEvent() { return this._lastEvent; }

        /* Get all rows as JSON string */
        getData() { return JSON.stringify(this._rows); }

        /* Replace all rows (called from script after loadData fires) */
        setData(raw) {
            try {
                var p = typeof raw === "string" ? JSON.parse(raw) : raw;
                this._rows = Array.isArray(p) ? p : [];
            } catch (e) { this._rows = []; }
            this._valid = null;
            this._dd = null; this._ddSearch = "";
            this._buildBody();
            this._updateStatus();
        }

        /* Set one cell value */
        setCellValue(ri, field, val) {
            if (!this._rows[ri]) { return; }
            this._rows[ri][field]     = val;
            this._rows[ri].isModified = true;
            this._valid = null;
            this._buildBody();
            this._updateStatus();
        }

        /* Set dropdown options for one cell */
        setRowOptions(ri, field, raw) {
            if (!this._opts[ri]) { this._opts[ri] = {}; }
            try {
                this._opts[ri][field] = typeof raw === "string" ? JSON.parse(raw) : raw;
            } catch (e) {
                this._opts[ri][field] = [];
            }
            this._buildBody();
        }
    }

    if (!customElements.get("com-madhav-cwipassetentry")) {
        customElements.define("com-madhav-cwipassetentry", CWIPAssetEntry);
    }

})();
