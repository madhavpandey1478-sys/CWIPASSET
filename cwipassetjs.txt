var template = document.createElement("template");

class CWIPAssetEntry extends HTMLElement {
    constructor() {
        super();

        this.attachShadow({mode:"open"})
            .appendChild(template.content.cloneNode(true));

        this._rows = [];
        this._selected = new Set();
        this._options = {assetClass:[], companyCode:[]};
        this._rowOptions = {};
        this._page = 1;
        this._pageSize = 50;
        this._search = "";
        this._classFilter = "";
        this._companyCodeFilter = "";
        this._lastEvent = "";
        this._valid = null;
        this._errors = [];
        this._openDropdown = null;

        var self = this;

        [
            "search","visibleCount","classFilter","companyCodeFilter",
            "load","validate","save","clear",
            "range","prev","page","next",
            "head","body","total","selected","valid","errors"
        ].forEach(function(id) {
            self[id] = self.shadowRoot.getElementById(id);
        });

        this._bind();
        this._render();
    }

    connectedCallback() {
        this._fire("onReady","ready");
    }

    _bind() {
        var self = this;

        this.search.addEventListener("input",function() {
            self._search = this.value;
            self._page = 1;
            self._render();
        });

        this.classFilter.addEventListener("change",function() {
            self._classFilter = this.value;
            self._page = 1;
            self._render();
        });

        this.companyCodeFilter.addEventListener("change",function() {
            self._companyCodeFilter = this.value;
            self._page = 1;
            self._render();
        });

        this.prev.onclick = function() {
            if (self._page > 1) {
                self._page--;
                self._render();
            }
        };

        this.next.onclick = function() {
            if (self._page < self._pageCount()) {
                self._page++;
                self._render();
            }
        };

        this.load.onclick = function() {
            self._fire("onDataChange","loadData");
        };

        this.validate.onclick = function() {
            self._validate();
        };

        this.save.onclick = function() {
            self._fire("onDataChange","saveChanges|" + self._rows.length);
        };

        this.clear.onclick = function() {
            self._clearSelectedOrAll();
        };

        this.shadowRoot.addEventListener("click",function(e) {
            if (!e.target.closest(".dropdown-wrap")) {
                if (self._openDropdown) {
                    self._openDropdown = null;
                    self._render();
                }
            }
        });
    }

    _fire(name,value) {
        this._lastEvent = String(value == null ? "" : value);
        this.dispatchEvent(new Event(name));
    }

    _key(row,index) {
        if (!row.__widgetKey) {
            row.__widgetKey = "asset_" + index + "_" +
                Math.random().toString(36).slice(2);
        }
        return row.__widgetKey;
    }

    _normalize(rows) {
        return rows.map(function(source) {
            var row = Object.assign({},source);

            if (row.assetId === undefined && row.AssetID !== undefined)
                row.assetId = row.AssetID;

            if (row.assetClass === undefined && row.AssetClass !== undefined)
                row.assetClass = row.AssetClass;

            if (row.companyCode === undefined && row.CompanyCode !== undefined)
                row.companyCode = row.CompanyCode;
            if (row.companyCode === undefined && row.company_code !== undefined)
                row.companyCode = row.company_code;
            if (row.companyCode == null) row.companyCode = "";

            if (row.capitalizedMonth === undefined &&
                row.CapitalizedMonth !== undefined)
                row.capitalizedMonth = row.CapitalizedMonth;

            if (row.assetId == null) row.assetId = "";
            if (row.assetClass == null) row.assetClass = "";
            if (row.capitalizedMonth == null) row.capitalizedMonth = "";

            row.selected = false;
            return row;
        });
    }

    /* Set rows from SAC scripting.
       Expected fields: assetId, assetClass, capitalizedMonth.
       capitalizedMonth should be YYYY-MM for input type="month". */
    setData(raw) {
        try {
            var parsed = typeof raw === "string" ? JSON.parse(raw) : raw;
            this._rows = this._normalize(Array.isArray(parsed) ? parsed : []);
        } catch(e) {
            this._rows = [];
        }

        this._selected.clear();
        this._rowOptions = {};
        this._page = 1;
        this._valid = null;
        this._errors = [];
        this._render();
    }

    /* Return all current rows for SAC scripting. */
    getData() {
        return JSON.stringify(this._rows.map(function(row) {
            var result = Object.assign({},row);
            delete result.__widgetKey;
            delete result.selected;
            return result;
        }));
    }

    getLastEvent() {
        return this._lastEvent;
    }

    getSelectedData() {
        var self = this;

        return JSON.stringify(this._rows.filter(function(row,index) {
            return self._selected.has(self._key(row,index));
        }).map(function(row) {
            var result = Object.assign({},row);
            delete result.__widgetKey;
            delete result.selected;
            return result;
        }));
    }

    setCellValue(rowIndex,field,value) {
        if (!this._rows[rowIndex]) return;

        /* Asset ID is always read-only in the UI. */
        if (field === "assetId") {
            this._rows[rowIndex].assetId = value;
        } else if (
            field === "assetClass" ||
            field === "capitalizedMonth"
        ) {
            this._rows[rowIndex][field] = value;
        } else {
            return;
        }

        this._valid = null;
        this._render();
    }

    /* Pass a JSON array of asset-class dropdown values. */
    setOptions(raw) {
        try {
            var parsed = typeof raw === "string" ? JSON.parse(raw) : raw;
            this._options.assetClass = Array.isArray(parsed) ? parsed : [];
        } catch(e) {
            this._options.assetClass = [];
        }

        this._syncClassFilter();
        this._render();
    }

    setRowOptions(rowIndex,field,raw) {
        if (field !== "assetClass" || !Number.isInteger(rowIndex) || rowIndex < 0) {
            return;
        }

        var options;
        try {
            options = typeof raw === "string" ? JSON.parse(raw) : raw;
        } catch (e) {
            options = [];
        }

        if (!Array.isArray(options)) {
            options = [];
        }

        if (!this._rowOptions[rowIndex]) {
            this._rowOptions[rowIndex] = {};
        }

        this._rowOptions[rowIndex][field] = options;

        if (this._openDropdown === rowIndex) {
            this._openDropdown = null;
        }

        this._render();
    }

    _syncClassFilter() {
        var self = this;
        var oldValue = this._classFilter;
        var select = this.classFilter;

        select.innerHTML = '<option value="">All Classes</option>';

        (this._options.assetClass || []).forEach(function(option) {
            var value = typeof option === "object" ? option.value : option;
            var label = typeof option === "object" ?
                (option.text !== undefined ? option.text : option.value) :
                option;

            var opt = document.createElement("option");
            opt.value = String(value);
            opt.textContent = String(label);
            select.appendChild(opt);
        });

        select.value = oldValue;
        if (select.value !== oldValue) this._classFilter = "";
    }

    setCompanyCodeOptions(raw) {
        try {
            var parsed = typeof raw === "string" ? JSON.parse(raw) : raw;
            this._options.companyCode = Array.isArray(parsed) ? parsed : [];
        } catch(e) {
            this._options.companyCode = [];
        }
        this._syncCompanyCodeFilter();
        this._render();
    }

    _syncCompanyCodeFilter() {
        var select = this.companyCodeFilter;
        var oldValue = this._companyCodeFilter;
        var configured = this._options.companyCode || [];
        var values = configured.length ? configured : this._rows.map(function(row) { return row.companyCode; });
        var unique = [];
        values.forEach(function(option) {
            var value = typeof option === "object" ? option.value : option;
            if (value !== undefined && value !== null && String(value) !== "" && !unique.some(function(v) { return String(v.value) === String(value); })) {
                unique.push({value:value, text: typeof option === "object" && option.text !== undefined ? option.text : value});
            }
        });
        select.innerHTML = '<option value="">All Company Codes</option>';
        unique.forEach(function(item) {
            var opt = document.createElement("option");
            opt.value = String(item.value);
            opt.textContent = String(item.text);
            select.appendChild(opt);
        });
        select.value = oldValue;
        if (select.value !== oldValue) this._companyCodeFilter = "";
    }

    _filtered() {
        var query = this._search.toLowerCase();
        var assetClass = this._classFilter;
        var companyCode = this._companyCodeFilter;

        return this._rows.map(function(row,index) {
            return {row:row,index:index};
        }).filter(function(item) {
            var id = String(item.row.assetId || "").toLowerCase();
            var cls = String(item.row.assetClass || "");
            var code = String(item.row.companyCode || "");

            return (!query || id.indexOf(query) !== -1) &&
                   (!assetClass || cls === assetClass) &&
                   (!companyCode || code === companyCode);
        });
    }

    _pageCount() {
        return Math.max(1,Math.ceil(this._filtered().length / this._pageSize));
    }

    _render() {
        var self = this;
        var filtered = this._filtered();
        var pageCount = this._pageCount();

        if (this._page > pageCount) this._page = pageCount;

        var start = (this._page - 1) * this._pageSize;
        var pageItems = filtered.slice(start,start + this._pageSize);

        this._syncClassFilter();
        this._syncCompanyCodeFilter();

        this.head.innerHTML =
            '<th class="selhead">Sel <input id="selectAll" class="check" type="checkbox"></th>' +
            FIELDS.map(function(col) {
                return '<th style="width:' + col.width + 'px">' +
                    self._esc(col.label) + '</th>';
            }).join("");

        var selectAll = this.shadowRoot.getElementById("selectAll");

        selectAll.checked = pageItems.length > 0 &&
            pageItems.every(function(item) {
                return self._selected.has(self._key(item.row,item.index));
            });

        selectAll.onchange = function() {
            pageItems.forEach(function(item) {
                var key = self._key(item.row,item.index);

                if (selectAll.checked) self._selected.add(key);
                else self._selected.delete(key);
            });

            self._render();
            self._fire("onDataChange","selectAll|" + selectAll.checked);
        };

        this.body.innerHTML = "";

        if (!pageItems.length) {
            this.body.innerHTML =
                '<tr><td class="empty" colspan="4">No matching records</td></tr>';
        }

        pageItems.forEach(function(item) {
            var row = item.row;
            var index = item.index;
            var key = self._key(row,index);
            var tr = document.createElement("tr");

            if (self._selected.has(key)) tr.classList.add("selected");

            var selectCell = document.createElement("td");
            var checkbox = document.createElement("input");

            checkbox.type = "checkbox";
            checkbox.className = "check";
            checkbox.checked = self._selected.has(key);

            checkbox.onchange = function() {
                if (checkbox.checked) self._selected.add(key);
                else self._selected.delete(key);

                self._render();
                self._fire("onDataChange","selectRow|" + index + "|" + checkbox.checked);
            };

            selectCell.appendChild(checkbox);
            tr.appendChild(selectCell);

            FIELDS.forEach(function(col) {
                var td = document.createElement("td");
                var control;

                if (col.type === "select") {
                    control = self._buildDropdown(row,index,col);
                } else {
                    control = document.createElement("input");
                    control.type = col.type;
                    control.className = "cell";
                    control.value = row[col.field] || "";

                    if (col.readonly) {
                        control.readOnly = true;
                        control.setAttribute("aria-label","Read-only Asset ID");
                    }

                    control.addEventListener("change",function() {
                        if (col.readonly) return;

                        row[col.field] = control.value;
                        self._valid = null;
                        self._errors = [];
                        self._updateStatus();

                        self._fire(
                            "onFieldChange",
                            "fieldChange|" + index + "|" +
                            col.field + "|" + control.value
                        );
                    });
                }

                td.appendChild(control);
                tr.appendChild(td);
            });

            self.body.appendChild(tr);
        });

        this.visibleCount.textContent = filtered.length;

        this.range.textContent = filtered.length ?
            "Showing " + (start + 1) + " to " +
            Math.min(start + this._pageSize,filtered.length) +
            " of " + filtered.length :
            "Showing 0 to 0 of 0";

        this.page.textContent = "Page " + this._page + " of " + pageCount;
        this.prev.disabled = this._page <= 1;
        this.next.disabled = this._page >= pageCount;

        this._updateStatus();
    }

    _buildDropdown(row,index,col) {
        var self = this;
        var wrap = document.createElement("div");
        wrap.className = "dropdown-wrap";

        var value = row[col.field] || "";
        var rowSpecific = this._rowOptions[index] &&
            this._rowOptions[index][col.field];
        var options = rowSpecific || this._options.assetClass || [];
        var open = this._openDropdown === index;

        var selectedOption = options.find(function(option) {
            var v = typeof option === "object" ? option.value : option;
            return String(v) === String(value);
        });

        var label = selectedOption ?
            (typeof selectedOption === "object" ?
                (selectedOption.text !== undefined ?
                    selectedOption.text : selectedOption.value) :
                selectedOption) :
            value;

        var button = document.createElement("button");
        button.type = "button";
        button.className = "dropdown";

        var valueSpan = document.createElement("span");
        valueSpan.className = "dropdown-value" +
            (label ? "" : " dropdown-placeholder");
        valueSpan.textContent = label || "Select Asset Class";

        var arrow = document.createElement("span");
        arrow.textContent = open ? "▴" : "▾";

        button.appendChild(valueSpan);
        button.appendChild(arrow);

        button.onclick = function(e) {
            e.stopPropagation();
            self._openDropdown = open ? null : index;
            self._render();
        };

        wrap.appendChild(button);

        if (!open) return wrap;

        var panel = document.createElement("div");
        panel.className = "dropdown-panel";

        var search = document.createElement("input");
        search.className = "dropdown-search";
        search.placeholder = "Search asset class...";
        search.type = "text";

        var list = document.createElement("div");
        list.className = "option-list";

        panel.appendChild(search);
        panel.appendChild(list);
        wrap.appendChild(panel);

        function renderOptions(query) {
            list.innerHTML = "";
            var q = query.toLowerCase();

            var matches = options.filter(function(option) {
                var v = typeof option === "object" ? option.value : option;
                var text = typeof option === "object" ?
                    (option.text !== undefined ? option.text : option.value) :
                    option;

                return String(text).toLowerCase().indexOf(q) !== -1 ||
                       String(v).toLowerCase().indexOf(q) !== -1;
            });

            if (!matches.length) {
                var empty = document.createElement("div");
                empty.className = "no-options";
                empty.textContent = "No matching asset classes";
                list.appendChild(empty);
                return;
            }

            matches.forEach(function(option) {
                var v = typeof option === "object" ? option.value : option;
                var text = typeof option === "object" ?
                    (option.text !== undefined ? option.text : option.value) :
                    option;

                var opt = document.createElement("div");
                opt.className = "option" +
                    (String(v) === String(value) ? " active" : "");
                opt.textContent = String(text);

                opt.onclick = function(e) {
                    e.stopPropagation();

                    row.assetClass = String(v);
                    self._openDropdown = null;
                    self._valid = null;
                    self._errors = [];
                    self._render();

                    self._fire(
                        "onFieldChange",
                        "fieldChange|" + index + "|assetClass|" + v
                    );
                };

                list.appendChild(opt);
            });
        }

        renderOptions("");

        search.addEventListener("input",function() {
            renderOptions(search.value);
        });

        search.addEventListener("click",function(e) {
            e.stopPropagation();
        });

        return wrap;
    }

    _validate() {
        var errors = [];

        this._rows.forEach(function(row,index) {
            if (!row.assetId) {
                errors.push({row:index,field:"assetId",message:"Asset ID missing"});
            }

            if (!row.assetClass) {
                errors.push({row:index,field:"assetClass",message:"Asset Class required"});
            }

            if (!row.capitalizedMonth) {
                errors.push({
                    row:index,
                    field:"capitalizedMonth",
                    message:"Capitalized Month required"
                });
            } else if (!/^\d{4}-\d{2}$/.test(row.capitalizedMonth)) {
                errors.push({
                    row:index,
                    field:"capitalizedMonth",
                    message:"Month must be YYYY-MM"
                });
            }
        });

        this._errors = errors;
        this._valid = errors.length === 0;
        this._updateStatus();

        this._fire(
            "onValidate",
            (this._valid ? "VALID" : "INVALID") + "|" + errors.length
        );
    }

    _clearSelectedOrAll() {
        var self = this;

        if (this._selected.size) {
            this._rows.forEach(function(row,index) {
                if (self._selected.has(self._key(row,index))) {
                    /* Never clear the model-supplied Asset ID. */
                    row.assetClass = "";
                    row.capitalizedMonth = "";
                }
            });

            this._selected.clear();
        } else {
            /* With no selection, clear editable values for all rows. */
            this._rows.forEach(function(row) {
                row.assetClass = "";
                row.capitalizedMonth = "";
            });
        }

        this._valid = null;
        this._errors = [];
        this._openDropdown = null;
        this._render();
        this._fire("onClear","clear");
    }

    _updateStatus() {
        this.total.textContent = this._rows.length;
        this.selected.textContent = this._selected.size;
        this.valid.textContent =
            this._valid === null ? "-" :
            (this._valid ? "Valid" : "Invalid");
        this.errors.textContent = this._errors.length;
    }

    _esc(value) {
        return String(value == null ? "" : value)
            .replace(/&/g,"&amp;")
            .replace(/</g,"&lt;")
            .replace(/>/g,"&gt;")
            .replace(/"/g,"&quot;")
            .replace(/'/g,"&#039;");
    }
