        .sico svg {
            display:block;
        }
    `;

    /* =========================================================
       TEMPLATE
       ========================================================= */
    var tmpl = document.createElement("template");

    tmpl.innerHTML =
        '<style>' + STYLES + '</style>' +
        '<div class="container">' +
            '<div class="tab-bar" id="tabBar"></div>' +
            '<div class="toolbar" id="toolbar"></div>' +
            '<div class="table-area">' +
                '<table><thead><tr id="thead"></tr></thead>' +
                '<tbody id="tbody"></tbody></table>' +
            '</div>' +
            '<div class="status">' +
                '<span class="si">Total Rows: <span id="sRows" class="sv">0</span></span>' +
                '<span class="si">Selected Rows: <span id="sSel" class="sv">0</span></span>' +
                '<span class="si">Validation: <span id="sIco" class="sico"></span><span id="sVal" class="sv">-</span></span>' +
                '<span class="si">Error Rows: <span id="sErr" class="sv">0</span></span>' +
            '</div>' +
        '</div>';

    /* =========================================================
       COMPONENT
       ========================================================= */
    class CWIPAssetEntry extends HTMLElement {

        constructor() {
            super();

            this.attachShadow({ mode: "open" });
            this.shadowRoot.appendChild(tmpl.content.cloneNode(true));

            this._tab = "create";
            this._lastEvent = "";
            this._dd = null;
            this._ddSearch = "";

            this._data = {
                create: {
                    rows: [this._emptyRow()],
                    opts: {},
                    valid: null
                },
                modify: {
                    rows: [],
                    opts: {},
                    valid: null
                }
            };

            this._buildTabBar();
            this._render();
        }

        connectedCallback() {
            var self = this;

            this._globalClose = function () {
                if (self._dd) {
                    self._dd = null;
                    self._ddSearch = "";
                    self._buildBody();
                }
            };

            document.addEventListener("click", this._globalClose);
            this._fire("onReady", "ready");
        }

        disconnectedCallback() {
            if (this._globalClose) {
                document.removeEventListener("click", this._globalClose);
            }
        }

        /* ── helpers ── */

        _emptyRow() {
            var r = {
                selected: false,
                isModified: false
            };

            for (var i = 0; i < ALL_FIELDS.length; i++) {
                r[ALL_FIELDS[i]] = "";
            }

            return r;
        }

        _d() {
            return this._data[this._tab];
        }

        _cols() {
            return this._tab === "create" ? TAB1_COLS : TAB2_COLS;
        }

        _mand() {
            return this._tab === "create" ? TAB1_MANDATORY : TAB2_MANDATORY;
        }

        _pfx() {
            return this._tab === "modify" ? "2" : "";
        }

        _esc(v) {
            if (v == null) {
                return "";
            }

            return String(v)
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }

        _fire(name, value) {
            this._lastEvent = String(value);
            this.dispatchEvent(new Event(name));
        }

        /* ── option helpers ── */

        _oval(o) {
            return typeof o === "object" ? o.value : o;
        }

        _otxt(o) {
            return typeof o === "object"
                ? (o.text !== undefined ? o.text : o.value)
                : o;
        }

        _otext(opts, val) {
            if (!val) {
                return "";
            }

            for (var i = 0; i < opts.length; i++) {
                if (String(this._oval(opts[i])) === String(val)) {
                    return this._otxt(opts[i]);
                }
            }

            return String(val);
        }

        _ofilt(opts, q) {
            if (!q) {
                return opts;
            }

            var n = String(q).toLowerCase();
            var r = [];

            for (var i = 0; i < opts.length; i++) {
                if (
                    String(this._otxt(opts[i])).toLowerCase().indexOf(n) !== -1 ||
                    String(this._oval(opts[i])).toLowerCase().indexOf(n) !== -1
                ) {
                    r.push(opts[i]);
                }
            }

            return r;
        }

        _getOpts(tab, row, field) {
            var d = this._data[tab];

            return (d.opts[row] && d.opts[row][field])
                ? d.opts[row][field]
                : [];
        }

        /* ── validation helpers ── */

        _hasData(row) {
            for (var i = 0; i < ALL_FIELDS.length; i++) {
                var v = row[ALL_FIELDS[i]];

                if (v !== "" && v !== null && v !== undefined) {
                    return true;
                }
            }

            return false;
        }

        _missing(row, mand) {
            var m = [];

            for (var i = 0; i < mand.length; i++) {
                if (!row[mand[i]]) {
                    m.push(mand[i]);
                }
            }

            return m;
        }

        _errCount(d, mand) {
            var n = 0;

            for (var i = 0; i < d.rows.length; i++) {
                if (!this._hasData(d.rows[i])) {
                    continue;
                }

                if (this._missing(d.rows[i], mand).length > 0) {
                    n++;
                }
            }

            return n;
        }
                d.valid === false && this._hasData(row)
            ) ? this._missing(row, mand) : [];

            var tr = document.createElement("tr");

            if (row.selected) {
                tr.classList.add("sel");
            }

            if (miss.length) {
                tr.classList.add("err");
            }

            /* checkbox cell */
            var chkTd = document.createElement("td");
            chkTd.className = "chk-cell";

            var chk = document.createElement("input");
            chk.type = "checkbox";
            chk.className = "chk";
            chk.checked = !!row.selected;

            chk.onchange = (function (rowIdx, trEl) {
                return function (e) {
                    var chkd = e.target.checked;

                    d.rows[rowIdx].selected = chkd;

                    if (chkd) {
                        trEl.classList.add("sel");
                    } else {
                        trEl.classList.remove("sel");
                    }

                    self._updateDelBtn();
                    self._updateChkAll();
                    self._updateStatus();

                    self._fire(
                        "onDataChange",
                        "selectRow" +
                        (tab === "modify" ? "2" : "") +
                        "|" + rowIdx + "|" + chkd
                    );
                };
            }(ri, tr));

            chkTd.appendChild(chk);
            tr.appendChild(chkTd);

            /* data cells */
            for (var ci = 0; ci < cols.length; ci++) {
                var col = cols[ci];
                var hasErr = (miss.indexOf(col.field) !== -1);
                var td = document.createElement("td");

                if (col.type === "select") {
                    td.appendChild(
                        this._buildCombo(tab, d, ri, col, hasErr)
                    );
                } else {
                    td.appendChild(
                        this._buildInput(tab, d, ri, col, hasErr)
                    );
                }

                if (hasErr) {
                    var ep = document.createElement("div");
                    ep.className = "field-error";
                    ep.textContent = col.label + " is required";
                    td.appendChild(ep);
                }

                tr.appendChild(td);
            }

            return tr;
        }

        /* ── input cell ── */

        _buildInput(tab, d, ri, col, hasErr) {
            var self = this;
            var inp = document.createElement("input");

            inp.className =
                "cell" +
                (col.readonly ? " ro" : "") +
                (hasErr ? " ce" : "");

            inp.type = col.type || "text";
            inp.value = d.rows[ri][col.field] || "";

            if (col.readonly) {
                inp.readOnly = true;
            }

            if (!col.readonly) {
                inp.onchange = function () {
                    d.rows[ri][col.field] = inp.value;
                    d.rows[ri].isModified = true;
                    d.valid = null;

                    self._updateStatus();

                    self._fire(
                        "onDataChange",
                        "fieldChange" +
                        (tab === "modify" ? "2" : "") +
                        "|" + ri +
                        "|" + col.field +
                        "|" + inp.value
                    );
                };
            }

            return inp;
        }

        /* ── searchable dropdown cell ── */

        _buildCombo(tab, d, ri, col, hasErr) {
            var self = this;
            var field = col.field;
            var opts = this._getOpts(tab, ri, field);
            var curVal = d.rows[ri][field] || "";
            var curTxt = this._otext(opts, curVal) || "";

            var isOpen = !!(
                this._dd &&
                this._dd.tab === tab &&
                this._dd.row === ri &&
                this._dd.field === field
            );

            var wrap = document.createElement("div");
            wrap.className = "cw";
            wrap.setAttribute(
                "data-ct",
                tab + "|" + ri + "|" + field
            );

            /* dropdown toggle */
            var btn = document.createElement("button");
            btn.type = "button";
            btn.className =
                "ct" +
                (isOpen ? " open" : "") +
                (hasErr ? " ce" : "");

            var span = document.createElement("span");
            span.className = "ct-text" + (curTxt ? "" : " ph");
            span.textContent = curTxt || "Select";

            var chev = document.createElement("span");
            chev.className = "ct-chev";

            chev.innerHTML =
                '<svg width="12" height="12" viewBox="0 0 20 20" fill="none">' +
                '<path d="M5 7.5l5 5 5-5" stroke="currentColor" ' +
                'stroke-width="1.8" stroke-linecap="round" ' +
                'stroke-linejoin="round"/>' +
                '</svg>';

            btn.appendChild(span);
            btn.appendChild(chev);

            btn.onclick = function (e) {
                e.stopPropagation();

                if (isOpen) {
                    self._dd = null;
                } else {
                    self._dd = {
                        tab: tab,
                        row: ri,
                        field: field
                    };

                    self._ddSearch = "";
                }

                self._buildBody();
            };

            wrap.appendChild(btn);

            /* dropdown panel */
            if (isOpen) {
                var panel = document.createElement("div");
                panel.className = "cp";

                panel.onclick = function (e) {
                    e.stopPropagation();
                };

                /* search row */
                var sRow = document.createElement("div");
                sRow.className = "cs";

                var sIco = document.createElement("span");
                sIco.className = "cs-icon";

                sIco.innerHTML =
                    '<svg width="14" height="14" viewBox="0 0 20 20" fill="none">' +
                    '<circle cx="8.5" cy="8.5" r="5.5" stroke="currentColor" stroke-width="1.6"/>' +
                    '<path d="M13 13l4 4" stroke="currentColor" stroke-width="1.6" stroke-linecap="round"/>' +
                    '</svg>';

                var sInp = document.createElement("input");
                sInp.type = "text";
                sInp.className = "cs-inp";
                sInp.placeholder = "Search";
                sInp.value = this._ddSearch;
                sInp.autocomplete = "off";

                sRow.appendChild(sIco);
                sRow.appendChild(sInp);

                /* clear search button */
                if (this._ddSearch) {
                    var sClr = document.createElement("button");

                    sClr.type = "button";
                    sClr.className = "cs-clr";
                    sClr.innerHTML = "&times;";

                    sClr.onclick = function (e) {
                        e.stopPropagation();
                        self._ddSearch = "";
                        self._buildBody();
                    };

                    sRow.appendChild(sClr);
                }

                panel.appendChild(sRow);

                /* options list */
                var list = document.createElement("div");
                list.className = "co-list";

                var filtered = this._ofilt(opts, this._ddSearch);

                if (!filtered.length) {
                    var emp = document.createElement("div");
                    emp.className = "co-empty";
                    emp.textContent = "No matches";
                    list.appendChild(emp);

                } else {
                    for (var oi = 0; oi < filtered.length; oi++) {
                        var ov = this._oval(filtered[oi]);
                        var ot = this._otxt(filtered[oi]);

                        var item = document.createElement("div");

                        item.className =
                            "co-item" +
                            (String(ov) === String(curVal) ? " picked" : "");

                        item.textContent = ot;
                        item.setAttribute("data-v", ov);

                        item.onclick = (function (optVal) {
                            return function (e) {
                                e.stopPropagation();

                                d.rows[ri][field] = optVal;
                                d.rows[ri].isModified = true;
                                d.valid = null;

                                self._dd = null;
                                self._ddSearch = "";

                                self._buildBody();
                                self._updateStatus();

                                self._fire(
                                    "onDataChange",
                                    "fieldChange" +
                                    (tab === "modify" ? "2" : "") +
                                    "|" + ri +
                                    "|" + field +
                                    "|" + optVal
                                );
                            };
                        }(String(ov)));

                        list.appendChild(item);
                    }
                }

                panel.appendChild(list);
                wrap.appendChild(panel);

                /* focus and live search */
                setTimeout(function () {
                    sInp.focus();
                }, 0);

                sInp.oninput = function () {
                    self._ddSearch = sInp.value;
                    self._buildBody();
                };

                sInp.onkeydown = function (e) {
                    if (e.key === "Escape") {
                        self._dd = null;
                        self._ddSearch = "";
                        self._buildBody();
                    }
                };
            }

            return wrap;
        }

        /* ── status bar ── */

        _updateStatus() {
            var d = this._d();
            var mand = this._mand();
            var sel = 0;

            for (var i = 0; i < d.rows.length; i++) {
                if (d.rows[i].selected) {
                    sel++;
                }
            }

            this.shadowRoot.getElementById("sRows").textContent =
                d.rows.length;

            this.shadowRoot.getElementById("sSel").textContent = sel;

            var valEl = this.shadowRoot.getElementById("sVal");
            var ico = this.shadowRoot.getElementById("sIco");
            var errEl = this.shadowRoot.getElementById("sErr");

            valEl.className = "sv";

            if (d.valid === true) {
                valEl.textContent = "true";
                valEl.classList.add("ok");
                errEl.textContent = "0";

                ico.innerHTML =
                    '<svg width="12" height="12" viewBox="0 0 20 20" fill="none">' +
                    '<circle cx="10" cy="10" r="9" stroke="#107e3e" stroke-width="1.6"/>' +
                    '<path d="M6 10.3l2.6 2.6L14 7.5" stroke="#107e3e" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/>' +
                    '</svg>';

            } else if (d.valid === false) {
                valEl.textContent = "false";
                valEl.classList.add("bad");
                errEl.textContent = String(this._errCount(d, mand));

                ico.innerHTML =
                    '<svg width="12" height="12" viewBox="0 0 20 20" fill="none">' +
                    '<circle cx="10" cy="10" r="9" stroke="#bb0000" stroke-width="1.6"/>' +
                    '<path d="M10 6v5" stroke="#bb0000" stroke-width="1.8" stroke-linecap="round"/>' +
                    '<circle cx="10" cy="13.6" r="1" fill="#bb0000"/>' +
                    '</svg>';

            } else {
                valEl.textContent = "-";
                errEl.textContent = "0";
                ico.innerHTML = "";
            }
        }

        _updateDelBtn() {
            var d = this._d();
            var sel = 0;

            for (var i = 0; i < d.rows.length; i++) {
                if (d.rows[i].selected) {
                    sel++;
                }
            }

            var id = this._tab === "create" ? "btnDel" : "btnDel2";
            var el = this.shadowRoot.getElementById(id);

            if (el) {
                el.style.display = sel > 0 ? "inline-flex" : "none";
            }
        }

        _updateChkAll() {
            var d = this._d();
            var sel = 0;

            for (var i = 0; i < d.rows.length; i++) {
                if (d.rows[i].selected) {
                    sel++;
                }
            }

            var ca = this.shadowRoot.getElementById("chkAll");

            if (!ca) {
                return;
            }

            ca.checked = (d.rows.length > 0 && sel === d.rows.length);
            ca.indeterminate = (sel > 0 && sel < d.rows.length);
        }
        /* =========================================================
           PUBLIC API
           ========================================================= */

        getLastEvent() {
            return this._lastEvent;
        }

        getActiveTab() {
            return this._tab;
        }

        setActiveTab(t) {
            if (t !== "create" && t !== "modify") {
                return;
            }

            if (t === this._tab) {
                return;
            }

            this._tab = t;
            this._dd = null;
            this._ddSearch = "";

            var all = this.shadowRoot.querySelectorAll(".tab-btn");

            for (var i = 0; i < all.length; i++) {
                all[i].classList.toggle(
                    "active",
                    all[i].getAttribute("data-t") === t
                );
            }

            this._render();
        }

        /* ── Tab 1 API ── */

        getData() {
            return JSON.stringify(this._data.create.rows);
        }

        setData(v) {
            this._loadData("create", v, true);
        }

        setCellValue(r, f, v) {
            this._setCell("create", r, f, v);
        }

        setRowOptions(r, f, o) {
            this._setOpts("create", r, f, o);
        }

        /* ── Tab 2 API ── */

        getData2() {
            return JSON.stringify(this._data.modify.rows);
        }

        setData2(v) {
            this._loadData("modify", v, false);
        }

        setCellValue2(r, f, v) {
            this._setCell("modify", r, f, v);
        }

        setRowOptions2(r, f, o) {
            this._setOpts("modify", r, f, o);
        }

        /* ── shared internals ── */

        _loadData(tab, raw, forceOne) {
            var d = this._data[tab];

            try {
                var p = typeof raw === "string" ? JSON.parse(raw) : raw;
                d.rows = Array.isArray(p) ? p : [];
            } catch (e) {
                d.rows = [];
            }

            if (forceOne && !d.rows.length) {
                d.rows.push(this._emptyRow());
            }

            d.valid = null;

            if (this._dd && this._dd.tab === tab) {
                this._dd = null;
                this._ddSearch = "";
            }

            if (this._tab === tab) {
                this._render();
            }
        }

        _setCell(tab, ri, field, val) {
            var d = this._data[tab];

            if (!d.rows[ri]) {
                return;
            }

            d.rows[ri][field] = val;
            d.rows[ri].isModified = true;
            d.valid = null;

            if (this._tab === tab) {
                this._render();
            }
        }

        _setOpts(tab, ri, field, raw) {
            var d = this._data[tab];

            if (!d.opts[ri]) {
                d.opts[ri] = {};
            }

            try {
                d.opts[ri][field] =
                    typeof raw === "string" ? JSON.parse(raw) : raw;
            } catch (e) {
                d.opts[ri][field] = [];
            }

            if (this._tab === tab) {
                this._buildBody();
            }
        }
    }

    if (!customElements.get("com-madhav-cwipassetentry")) {
        customElements.define(
            "com-madhav-cwipassetentry",
            CWIPAssetEntry
        );
    }

})();
